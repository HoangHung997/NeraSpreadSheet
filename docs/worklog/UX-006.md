## Đo layout: dữ liệu và giới hạn

Base `2bc00eb667da2f2c5afda1024ab753ac638d85d4` so với presentation code
`f9fc37247f3c7193c97d5a4344a7d16ff75382e0`. Các fix native sau đó không đổi
Commands/Ribbon.Core. Tái sử dụng nguyên Setup và method
`RibbonLayoutBenchmarks.PackAndCollapseSevenHundredTwentyCommands`: 9 tabs,
8 groups/tab, 10 commands/group; width 1536/1280/1024/820, scale 1, selected tab0.
Projection/localization nằm trong Setup, **không** nằm trong phép đo packing.
Release net10.0, SDK 10.0.302; lượt lưu raw xác minh runtime 10.0.10.
Không thêm package. Wrapper Stopwatch + GC.GetAllocatedBytesForCurrentThread,
1500 warm-up/width/variant, không dùng BenchmarkDotNet runner.

T0/T1: 7 samples × 2000 calls; T2/T3: 14 samples × 1000 calls/variant,
trong một process, hai AssemblyLoadContext, đảo trước/sau ở mỗi sample.
T2 đã được khởi chạy khi coordinator yêu cầu giữ raw; wrapper lúc đó chỉ lưu
summary. T3 giữ nguyên cấu hình T2 và bổ sung raw + kiểm tra snapshot. Các lượt
trước chỉ có summary, không thể khôi phục raw; toàn bộ summary được giữ bên dưới,
không loại lượt chậm hoặc trộn các cấu hình. Đơn vị latency là µs/op, allocation B/op.

CPU/desktop có công việc song song chưa kiểm soát; đây là yếu tố giới hạn, không
phải nguyên nhân latency đã được chứng minh. Các ranges rộng, có lượt chậm hơn
sau thay đổi; **chưa chấp nhận kết luận không hồi quy latency hoặc tăng tốc**.

```csharp
using System.Diagnostics;
using System.Reflection;
using System.Runtime.Loader;
using System.Text.Json;
using System.Security.Cryptography;

var results = new List<object>();
var before = Load(args[0]);
var after = Load(args[1]);
foreach (var width in new double[] { 1536, 1280, 1024, 820 })
{
    var operations = new[] { Create(before, width), Create(after, width) };
    var inputJson = operations.Select(operation =>
    {
        var target = operation.Target!;
        var snapshot = target.GetType().GetField("_presentation", BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(target)!;
        return JsonSerializer.Serialize(snapshot, snapshot.GetType());
    }).ToArray();
    var outputJson = operations.Select(operation => { var output = operation(); return JsonSerializer.Serialize(output, output.GetType()); }).ToArray();
    if (inputJson[0] != inputJson[1] || outputJson[0] != outputJson[1]) throw new InvalidOperationException("Paired snapshots/layout outputs differ.");
    var inputHash = Convert.ToHexString(SHA256.HashData(System.Text.Encoding.UTF8.GetBytes(inputJson[0])));
    var outputHash = Convert.ToHexString(SHA256.HashData(System.Text.Encoding.UTF8.GetBytes(outputJson[0])));
    var input = JsonDocument.Parse(inputJson[0]);
    var tabs = input.RootElement.GetProperty("Tabs").GetArrayLength();
    var commands = input.RootElement.GetProperty("Tabs").EnumerateArray().Sum(tab => tab.GetProperty("Groups").EnumerateArray().Sum(group => group.GetProperty("Items").GetArrayLength()));
    if (tabs != 9 || commands != 720) throw new InvalidOperationException("Workload size changed.");
    foreach (var operation in operations)
        for (var index = 0; index < 1500; index++) GC.KeepAlive(operation());
    var times = new[] { new List<double>(), new List<double>() };
    var bytes = new[] { new List<double>(), new List<double>() };
    for (var round = 0; round < 14; round++)
    {
        foreach (var side in round % 2 == 0 ? new[] { 0, 1 } : new[] { 1, 0 })
        {
            GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
            var operation = operations[side];
            var allocated = GC.GetAllocatedBytesForCurrentThread();
            var start = Stopwatch.GetTimestamp();
            for (var iteration = 0; iteration < 1000; iteration++) GC.KeepAlive(operation());
            var elapsed = Stopwatch.GetElapsedTime(start).TotalMicroseconds / 1000;
            var allocation = (GC.GetAllocatedBytesForCurrentThread() - allocated) / 1000d;
            times[side].Add(elapsed); bytes[side].Add(allocation);
        }
    }
    foreach (var side in new[] { 0, 1 })
    {
        var rawMicroseconds = times[side].ToArray();
        var rawAllocatedBytes = bytes[side].ToArray();
        times[side].Sort(); bytes[side].Sort();
        results.Add(new { width, side = side == 0 ? "before" : "after", medianMicroseconds = (times[side][6] + times[side][7]) / 2,
            minimumMicroseconds = times[side][0], maximumMicroseconds = times[side][13], allocatedBytesPerOperation = (bytes[side][6] + bytes[side][7]) / 2,
            runtime = Environment.Version.ToString(), tabs, commands, scale = 1, inputHash, outputHash, rawMicroseconds, rawAllocatedBytes });
    }
}
Console.WriteLine(JsonSerializer.Serialize(results));

static Type Load(string path)
{
    var fullPath = Path.GetFullPath(path);
    var directory = Path.GetDirectoryName(fullPath)!;
    var context = new AssemblyLoadContext(fullPath);
    context.Resolving += (loader, name) => File.Exists(Path.Combine(directory, name.Name + ".dll"))
        ? loader.LoadFromAssemblyPath(Path.Combine(directory, name.Name + ".dll")) : null;
    return context.LoadFromAssemblyPath(fullPath).GetType("NeraSpreadSheet.Benchmarks.RibbonLayoutBenchmarks")!;
}

static Func<object> Create(Type type, double width)
{
    var benchmark = Activator.CreateInstance(type)!;
    type.GetProperty("Width")!.SetValue(benchmark, width);
    type.GetMethod("Setup")!.Invoke(benchmark, null);
    return type.GetMethod("PackAndCollapseSevenHundredTwentyCommands")!.CreateDelegate<Func<object>>(benchmark);
}
```
